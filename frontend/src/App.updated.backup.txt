import { Toaster, toast } from "sonner";
import { Link, NavLink, Route, Routes, useLocation, useNavigate } from "react-router-dom";
import Home from "./pages/Home";
import Packages from "./pages/Packages";
import travelMateLogo from "./assets/travelmate-logo.png";

function App() {
  const navigate = useNavigate();
  const location = useLocation();
  const [isLoggedIn, setIsLoggedIn] = useState(() => Boolean(localStorage.getItem("token")));
  const [showBooking, setShowBooking] = useState(false);
  const [bookingConfirmed, setBookingConfirmed] = useState(false);
  const [confirmedDestination, setConfirmedDestination] = useState("");
  const [numberOfPeople, setNumberOfPeople] = useState("");
  const [travelDate, setTravelDate] = useState("");
  const [bookingSubmitting, setBookingSubmitting] = useState(false);
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");
  const [loginEmail, setLoginEmail] = useState("");
  const [loginPassword, setLoginPassword] = useState("");
  const [registerConfirmed, setRegisterConfirmed] = useState(false);
  const [myBookings, setMyBookings] = useState([]);
  const [bookingsLoading, setBookingsLoading] = useState(false);
  const [bookingsError, setBookingsError] = useState("");
  const [cancellingBookingId, setCancellingBookingId] = useState(null);
  const [bookingToCancel, setBookingToCancel] = useState(null);

  // The URL determines which package details to show, even after a refresh.
  const match = location.pathname.match(/^\/packages\/([^/]+)\/?$/);
  const packageSlug = match ? decodeURIComponent(match[1]).toLowerCase() : null;
  const activePackage = travelPackages.find(
    (item) => item.name.toLowerCase() === packageSlug
  );

  const openPackage = (packageName) => {
    setShowBooking(false);
    setBookingConfirmed(false);
    navigate(`/packages/${encodeURIComponent(packageName.toLowerCase())}`);
  };

  useEffect(() => {
    window.scrollTo(0, 0);
  }, [location.pathname]);

  useEffect(() => {
    if (showBooking) {
      document.getElementById("booking")?.scrollIntoView({ behavior: "smooth" });
    }
  }, [showBooking]);

  const fetchMyBookings = async () => {
    const token = localStorage.getItem("token");
    if (!token) {
      setBookingsError("Please log in to view your bookings.");
      setMyBookings([]);
      return;
    }
    setBookingsLoading(true);
    setBookingsError("");
    try {
      const response = await fetch(`${API_URL}/api/bookings`, {
        headers: { Authorization: `Bearer ${token}` },
      });
      const data = await response.json();
      if (!response.ok) throw new Error(data.message || "Could not load bookings");
      setMyBookings(data);
    } catch (error) {
      setBookingsError(error.message);
      setMyBookings([]);
    } finally {
      setBookingsLoading(false);
    }
  };

  useEffect(() => {
    if (location.pathname === "/my-bookings") {
      fetchMyBookings();
    }
  }, [location.pathname, isLoggedIn]);

  const handleBooking = async (event) => {
    event.preventDefault();
    const token = localStorage.getItem("token");
    if (!token) {
      toast.error("Please log in before booking.");
      navigate("/login", { state: { from: location.pathname } });
      return;
    }
    if (!activePackage) {
      toast.error("Please select a travel package.");
      return;
    }
    if (!travelDate || Number(numberOfPeople) < 1) {
      toast.error("Invalid booking details", {
        description: "Please select a travel date and enter at least one traveler.",
      });
      return;
    }
    setBookingSubmitting(true);
    try {
      const response = await fetch(`${API_URL}/api/bookings`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          Authorization: `Bearer ${token}`,
        },
        body: JSON.stringify({
          destination: activePackage.name,
          travelDate,
          numberOfPeople: Number(numberOfPeople),
        }),
      });
      const data = await response.json();
      if (!response.ok) {
        toast.error("Booking failed", { description: data.message || "Please try again later." });
        return;
      }
      setConfirmedDestination(activePackage.name);
      setShowBooking(false);
      setBookingConfirmed(true);
      setNumberOfPeople("");
      setTravelDate("");
    } catch (error) {
      console.error("Booking error:", error);
      toast.error("Unable to complete booking", {
        description: "Please check your connection and try again.",
      });
    } finally {
      setBookingSubmitting(false);
    }
  };

  const handleLogin = async (event) => {
    event.preventDefault();
    try {
      const response = await fetch(`${API_URL}/api/auth/login`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email: loginEmail, password: loginPassword }),
      });
      const data = await response.json();
      if (!response.ok) {
        toast.error("Login failed", { description: data.message || "Please try again." });
        return;
      }
      localStorage.setItem("token", data.token);
      setIsLoggedIn(true);
      setLoginPassword("");
      setMyBookings([]);
      toast.success(`Welcome back, ${data.user.name}!`, {
        description: "You have successfully logged in to TravelMate.",
      });
      navigate(location.state?.from || "/packages", { replace: true });
    } catch (error) {
      console.error("Login error:", error);
      toast.error("Login failed", { description: error.message || "Please try again." });
    }
  };

  const handleRegister = async (event) => {
    event.preventDefault();
    if (password !== confirmPassword) {
      toast.error("Passwords do not match!", {
        description: "Please enter the same password in both fields.",
      });
      return;
    }
    try {
      const response = await fetch(`${API_URL}/api/auth/register`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ name, email, password }),
      });
      const data = await response.json();
      if (!response.ok) {
        toast.error("Registration failed", { description: data.message || "Please try again." });
        return;
      }
      setRegisterConfirmed(true);
      setName("");
      setEmail("");
      setPassword("");
      setConfirmPassword("");
    } catch (error) {
      console.error("Registration error:", error);
      toast.error("Unable to register", { description: "Please check your connection and try again." });
    }
  };

  const handleCancelBooking = async (bookingId) => {
    const token = localStorage.getItem("token");
    if (!token) {
      toast.error("Please log in to cancel a booking.");
      return;
    }
    setCancellingBookingId(bookingId);
    try {
      const response = await fetch(`${API_URL}/api/bookings/${bookingId}/cancel`, {
        method: "PATCH",
        headers: { Authorization: `Bearer ${token}` },
      });
      const data = await response.json();
      if (!response.ok) throw new Error(data.message || "Could not cancel booking");
      setMyBookings((current) =>
        current.map((booking) =>
          booking._id === bookingId ? { ...booking, status: "Cancelled" } : booking
        )
      );
      toast.success("Booking cancelled successfully!", {
        description: "Your booking status has been updated.",
      });
    } catch (error) {
      toast.error("Unable to cancel booking", {
        description: error.message || "Please try again later.",
      });
    } finally {
      setCancellingBookingId(null);
    }
  };

  const handleLogout = () => {
    localStorage.removeItem("token");
    setIsLoggedIn(false);
    setMyBookings([]);
    setShowBooking(false);
    setBookingConfirmed(false);
    navigate("/");
    toast.success("Logged out successfully!", {
      description: "Thank you for using TravelMate. See you again!",
    });
  };

  return (
    <div>
      <Toaster position="top-right" richColors closeButton duration={4000} />
      <header>
        <Link to="/" className="travelmate-logo" aria-label="TravelMate home">
          <img src={travelMateLogo} alt="TravelMate" className="travelmate-logo-image" />
        </Link>
        <nav>
          <NavLink to="/">Home</NavLink>
          <NavLink to="/packages">Packages</NavLink>
          <NavLink to="/about">About</NavLink>
          {isLoggedIn ? (
            <>
              <NavLink to="/my-bookings">My Bookings</NavLink>
              <button type="button" onClick={handleLogout}>Logout</button>
            </>
          ) : (
            <>
              <NavLink to="/login">Login</NavLink>
              <NavLink to="/register">Register</NavLink>
            </>
          )}
        </nav>
      </header>

      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/packages" element={<Packages onSelectPackage={openPackage} />} />
        <Route
          path="/packages/:packageSlug"
          element={
            activePackage ? (
              <main>
                <section className="package-details">
                  <h2>{activePackage.name} Travel Package</h2>
                  <img
                    src={activePackage.image}
                    alt={`${activePackage.name} destination`}
                    style={{ width: "100%", maxWidth: 680, borderRadius: 12, display: "block", margin: "16px auto" }}
                  />
                  <p><strong>Duration:</strong> {activePackage.duration}</p>
                  <p><strong>Starting Price:</strong> ₹{activePackage.price.toLocaleString("en-IN")} per traveler</p>
                  <p>{activePackage.description}</p>
                  {!bookingConfirmed && (
                    <button className="book-button" type="button" onClick={() => {
                      if (!isLoggedIn) {
                        toast.error("Please log in before booking.");
                        navigate("/login", { state: { from: location.pathname } });
                        return;
                      }
                      setShowBooking(true);
                    }}>Book Now</button>
                  )}
                  <button className="close-button" type="button" onClick={() => {
                    setShowBooking(false);
                    setBookingConfirmed(false);
                    navigate("/packages");
                  }}>Back to Packages</button>
                </section>

                {showBooking && !bookingConfirmed && (
                  <section className="booking-section" id="booking">
                    <div className="booking-form">
                      <h2>Book Your Trip</h2>
                      <p>Booking for: <strong>{activePackage.name}</strong></p>
                      <form onSubmit={handleBooking}>
                        <label htmlFor="booking-name">Full Name</label>
                        <input id="booking-name" type="text" placeholder="Enter your name" required />
                        <label htmlFor="booking-email">Email</label>
                        <input id="booking-email" type="email" placeholder="Enter your email" required />
                        <label htmlFor="booking-people">Number of Travelers</label>
                        <input id="booking-people" type="number" min="1" placeholder="Number of travelers" value={numberOfPeople} onChange={(e) => setNumberOfPeople(e.target.value)} required />
                        <label htmlFor="booking-date">Travel Date</label>
                        <input id="booking-date" type="date" min={new Date().toLocaleDateString("en-CA")} value={travelDate} onChange={(e) => setTravelDate(e.target.value)} required />
                        <button type="submit" disabled={bookingSubmitting}>{bookingSubmitting ? "Saving Booking..." : "Confirm Booking"}</button>
                      </form>
                      <button className="close-button" type="button" onClick={() => setShowBooking(false)}>Cancel</button>
                    </div>
                  </section>
                )}

                {bookingConfirmed && (
                  <section className="confirmation-section">
                    <div className="confirmation-card">
                      <h2>🎉 Booking Request Submitted!</h2>
                      <p>Your <strong>{confirmedDestination}</strong> trip has been booked successfully.</p>
                      <p>Your booking status is Pending.</p>
                      <button type="button" onClick={() => {
                        setBookingConfirmed(false);
                        navigate("/packages");
                      }}>Back to Packages</button>
                      <button type="button" onClick={() => navigate("/my-bookings")}>My Bookings</button>
                    </div>
                  </section>
                )}
              </main>
            ) : (
              <main><section className="package-details"><h2>Package not found</h2><Link to="/packages">View all packages</Link></section></main>
            )
          }
        />
        <Route path="/about" element={
          <main><section className="about">
            <h2>About TravelMate</h2>
            <p>TravelMate helps you discover exciting destinations and plan memorable journeys with easy-to-explore travel packages.</p>
            <p>Choose a destination, explore the package details, and book your trip with just a few simple steps.</p>
          </section></main>
        } />
        <Route path="/login" element={
          <main><section className="login-section" id="login"><div className="login-form">
            <h2>Login to TravelMate</h2>
            <form onSubmit={handleLogin}>
              <label htmlFor="login-email">Email</label>
              <input id="login-email" type="email" placeholder="Enter your email" value={loginEmail} onChange={(e) => setLoginEmail(e.target.value)} required />
              <label htmlFor="login-password">Password</label>
              <input id="login-password" type="password" placeholder="Enter your password" value={loginPassword} onChange={(e) => setLoginPassword(e.target.value)} required />
              <button type="submit">Login</button>
            </form>
            <p className="switch-form">Don't have an account? <Link to="/register">Create one</Link></p>
            <button className="close-button" type="button" onClick={() => navigate("/")}>Close</button>
          </div></section></main>
        } />
        <Route path="/register" element={
          <main><section className="register-section" id="register"><div className="register-form">
            {registerConfirmed ? (
              <div className="register-confirmation-card">
                <h2>🎉 Account Created Successfully!</h2>
                <p>Welcome to <strong>TravelMate</strong>! Your account has been created successfully.</p>
                <button type="button" onClick={() => { setRegisterConfirmed(false); navigate("/login"); }}>Go to Login</button>
              </div>
            ) : (
              <>
                <h2>Create Your TravelMate Account</h2>
                <form onSubmit={handleRegister}>
                  <label htmlFor="register-name">Full Name</label>
                  <input id="register-name" type="text" placeholder="Enter your name" value={name} onChange={(e) => setName(e.target.value)} required />
                  <label htmlFor="register-email">Email</label>
                  <input id="register-email" type="email" placeholder="Enter your email" value={email} onChange={(e) => setEmail(e.target.value)} required />
                  <label htmlFor="register-password">Password</label>
                  <input id="register-password" type="password" placeholder="Create a password" value={password} onChange={(e) => setPassword(e.target.value)} required />
                  <label htmlFor="register-confirm">Confirm Password</label>
                  <input id="register-confirm" type="password" placeholder="Confirm your password" value={confirmPassword} onChange={(e) => setConfirmPassword(e.target.value)} required />
                  <button type="submit">Create Account</button>
                </form>
                <p className="switch-form">Already have an account? <Link to="/login">Login</Link></p>
                <button className="close-button" type="button" onClick={() => navigate("/")}>Close</button>
              </>
            )}
          </div></section></main>
        } />
        <Route path="/my-bookings" element={
          <main><section className="my-bookings-section" id="my-bookings">
            <h2>✈️ My Bookings</h2>
            {!isLoggedIn ? (
              <p>Please <Link to="/login" state={{ from: "/my-bookings" }}>log in</Link> to view your bookings.</p>
            ) : bookingsLoading ? (
              <p>Loading your bookings...</p>
            ) : bookingsError ? (
              <p>{bookingsError}</p>
            ) : myBookings.length === 0 ? (
              <p>No bookings found.</p>
            ) : myBookings.map((booking) => (
              <div key={booking._id} className="booking-card">
                <h3>📍 {booking.destination}</h3>
                <p>Travel Date: {new Date(booking.travelDate).toLocaleDateString("en-IN")}</p>
                <p>Travelers: {booking.numberOfPeople}</p>
                <p>Total Price: ₹{booking.totalPrice.toLocaleString("en-IN")}</p>
                <p>Status: {booking.status}</p>
                {booking.status !== "Cancelled" && (
                  <button type="button" onClick={() => setBookingToCancel(booking)} disabled={cancellingBookingId !== null} style={{ backgroundColor: "#dc2626", color: "white", marginTop: "10px" }}>
                    {cancellingBookingId === booking._id ? "Cancelling..." : "Cancel Booking"}
                  </button>
                )}
              </div>
            ))}
            {isLoggedIn && <button type="button" onClick={fetchMyBookings} disabled={bookingsLoading}>Refresh Bookings</button>}
            <button className="close-button" type="button" onClick={() => navigate("/")}>Close</button>
          </section></main>
        } />
        <Route path="*" element={<main><section className="about"><h2>Page not found</h2><Link to="/">Go Home</Link></section></main>} />
      </Routes>

      {bookingToCancel && (
        <div
          role="presentation"
          style={{
            position: "fixed", inset: 0, backgroundColor: "rgba(15, 23, 42, 0.65)",
            display: "flex", alignItems: "center", justifyContent: "center",
            zIndex: 9999, padding: "20px",
          }}
        >
          <div
            role="dialog"
            aria-modal="true"
            aria-labelledby="cancel-booking-title"
            style={{
              backgroundColor: "#ffffff", borderRadius: "16px", padding: "30px",
              maxWidth: "420px", width: "100%", textAlign: "center",
              boxShadow: "0 20px 60px rgba(0, 0, 0, 0.25)",
            }}
          >
            <h2 id="cancel-booking-title" style={{ color: "#1e293b" }}>Cancel Booking?</h2>
            <p style={{ color: "#64748b", margin: "16px 0 24px" }}>
              Are you sure you want to cancel your{" "}
              <strong>{bookingToCancel.destination}</strong> trip?
            </p>
            <div style={{ display: "flex", gap: "12px", justifyContent: "center" }}>
              <button
                type="button"
                onClick={() => setBookingToCancel(null)}
                disabled={cancellingBookingId !== null}
                style={{ backgroundColor: "#e2e8f0", color: "#334155", padding: "12px 20px", borderRadius: "8px" }}
              >
                Keep Booking
              </button>
              <button
                type="button"
                disabled={cancellingBookingId !== null}
                onClick={() => {
                  const bookingId = bookingToCancel._id;
                  setBookingToCancel(null);
                  handleCancelBooking(bookingId);
                }}
                style={{ backgroundColor: "#dc2626", color: "white", padding: "12px 20px", borderRadius: "8px" }}
              >
                Yes, Cancel
              </button>
            </div>
          </div>
        </div>
      )}

      <footer><p>© 2026 TravelMate. All rights reserved.</p></footer>
    </div>
  );
}

export default App;
